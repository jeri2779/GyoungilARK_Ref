// ─────────────────────────────────────────────────────────────
// Google Apps Script 웹앱 코드. Unity 프로젝트 컴파일 대상이 아님(.txt) —
// 스프레드시트 → 확장 프로그램 → Apps Script 편집기에 그대로 붙여넣는 참고용 파일.
//
// AnalyticsRecorder가 보내는 배치(JSON 배열, 각 원소는 evt 필드로 종류 구분)를 받아
// evt 이름과 같은 이름의 시트 탭에 한 줄씩 append한다. 탭이 없으면 자동 생성하고
// 첫 행에 헤더(필드 이름)를 만든다.
// ─────────────────────────────────────────────────────────────

// 배포 확인용. 브라우저로 exec URL을 열면(=GET 요청) "OK"가 뜨는지로 배포 자체가 살아있는지 확인한다.
// 편집기의 "실행" 버튼으로 이 함수를 직접 누르면 e가 undefined라 여전히 실패한다 — 반드시 브라우저나
// curl로 실제 URL에 접속해서 확인할 것.
function doGet(e) {
  return ContentService.createTextOutput("OK");
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var batch = JSON.parse(e.postData.contents);

  batch.forEach(function (record) {
    var sheetName = record.evt || "unknown";
    var sheet = ss.getSheetByName(sheetName);

    var keys = Object.keys(record);

    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(keys); // 첫 시트 생성 시 헤더 작성
    }

    // 시트가 이미 있는데 헤더가 비어있으면(수동 생성 등) 헤더를 채운다.
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(keys);
    }

    var header = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
    var row = header.map(function (col) { return record[col] !== undefined ? record[col] : ""; });
    sheet.appendRow(row);
  });

  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, count: batch.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
